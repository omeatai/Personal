Example

const images = ["nature", "snow", "mountains"];
let index = 0;
let timer;
function showNextSlide() {
  index = (index + 1) % images.length;
  document.getElementById("slide").textContent = images[index];
}
function startSlides() {
  if (!timer) {
    timer = setInterval(showNextSlide, 1000);
  }
}
