Example

const myObject = {
  data: ["A", "B", "C"],
  [Symbol.iterator]() {
    let index = 0;
    let data = this.data;
    return {
      next() {
        if (index < data.length) {
          return {value: data[index++], done: false};
        }
        return {done: true};
      }
    };
  }
};
let text = "";
for (const x of myObject) {
  text += x + " ";
}
